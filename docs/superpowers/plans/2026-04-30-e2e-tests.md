```python
import asyncio
import json
import os
import re
import shutil
import signal
import socket
import subprocess
import tempfile
import time
from pathlib import Path

import pytest

pytestmark = [
    pytest.mark.skipif(
        not shutil.which("direwolf"), reason="direwolf not installed"
    ),
    pytest.mark.skipif(
        not shutil.which("pw-loopback"), reason="pipewire not available"
    ),
]

needs_pat = pytest.mark.skipif(
    not shutil.which("pat"), reason="pat not installed"
)


def free_port():
    """Return a free TCP port number."""
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.bind(("127.0.0.1", 0))
        return s.getsockname()[1]


def wait_for_port(port, host="127.0.0.1", timeout=10.0):
    """Block until a TCP port is accepting connections."""
    deadline = time.monotonic() + timeout
    while time.monotonic() < deadline:
        try:
            with socket.create_connection((host, port), timeout=0.5):
                return True
        except OSError:
            time.sleep(0.1)
    raise TimeoutError(f"Port {port} not ready after {timeout}s")


def kill_proc(proc):
    """Terminate a subprocess, escalating to SIGKILL if needed."""
    if proc.poll() is not None:
        return
    proc.terminate()
    try:
        proc.wait(timeout=5)
    except subprocess.TimeoutExpired:
        proc.kill()
        proc.wait(timeout=5)
```

```python
def write_direwolf_config(
    path, mycall, adevice_playback, adevice_capture,
    agwport=0, kissport=0, kiss_pty=False,
):
    """Write a Direwolf configuration file.

    Args:
        path: Path to write the config file.
        mycall: Station callsign.
        adevice_playback: PipeWire sink node name for TX audio.
        adevice_capture: PipeWire source node name for RX audio.
        agwport: AGWPE TCP port (0 to disable).
        kissport: KISS TCP port (0 to disable).
        kiss_pty: If True, enable KISS pseudo-TTY (-p flag).
    """
    lines = [
        f"ADEVICE {adevice_capture} {adevice_playback}",
        "ACHANNELS 1",
        "ARATE 44100",
        f"MYCALL {mycall}",
        "MODEM 1200",
        f"AGWPORT {agwport}",
        f"KISSPORT {kissport}",
        "TXDELAY 10",
        "TXTAIL 5",
        "SLOTTIME 10",
        "PERSIST 255",
    ]
    Path(path).write_text("\n".join(lines) + "\n")


def parse_direwolf_pty(proc, timeout=10.0):
    """Read Direwolf stdout until the PTY path is found.

    Returns the PTY device path (e.g., /dev/pts/3).
    """
    deadline = time.monotonic() + timeout
    while time.monotonic() < deadline:
        line = proc.stdout.readline()
        if not line:
            time.sleep(0.1)
            continue
        line = line.decode(errors="replace").strip()
        m = re.search(r"Virtual KISS TNC is available on (/dev/pts/\d+)", line)
        if m:
            return m.group(1)
    raise TimeoutError("Direwolf did not report PTY path")


@pytest.fixture()
def direwolf_pair(pipewire_audio_link, tmp_path, request):
    """Start a pair of Direwolf instances connected via PipeWire audio.

    The fixture parameter 'kiss_mode' controls how Direwolf-A exposes KISS:
      - "tcp": KISS TCP port (default)
      - "pty": KISS pseudo-TTY

    Yields a dict with:
      - agwpe_port_a: AGWPE port for Direwolf-A (0 if disabled)
      - kiss_port_a: KISS TCP port for Direwolf-A (0 if PTY mode)
      - kiss_pty_a: PTY device path for Direwolf-A (None if TCP mode)
      - agwpe_port_b: AGWPE port for Direwolf-B
      - proc_a: Direwolf-A subprocess
      - proc_b: Direwolf-B subprocess
    """
    kiss_mode = getattr(request, "param", "tcp")
    audio = pipewire_audio_link

    agwpe_port_b = free_port()
    kiss_port_a = free_port() if kiss_mode == "tcp" else 0
    kiss_pty = kiss_mode == "pty"

    conf_a = tmp_path / "direwolf-a.conf"
    conf_b = tmp_path / "direwolf-b.conf"

    write_direwolf_config(
        conf_a, "N0CALL-1",
        adevice_playback=audio["a_playback"],
        adevice_capture=audio["a_capture"],
        agwport=0,
        kissport=kiss_port_a,
        kiss_pty=kiss_pty,
    )
    write_direwolf_config(
        conf_b, "N0CALL-2",
        adevice_playback=audio["b_playback"],
        adevice_capture=audio["b_capture"],
        agwport=agwpe_port_b,
        kissport=0,
    )

    # Start Direwolf-A
    cmd_a = ["direwolf", "-c", str(conf_a), "-t", "0"]
    if kiss_pty:
        cmd_a.append("-p")
    proc_a = subprocess.Popen(
        cmd_a,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
    )

    # Start Direwolf-B
    proc_b = subprocess.Popen(
        ["direwolf", "-c", str(conf_b), "-t", "0"],
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
    )

    try:
        # Wait for readiness
        pty_path = None
        if kiss_pty:
            pty_path = parse_direwolf_pty(proc_a)
        elif kiss_port_a:
            wait_for_port(kiss_port_a)

        wait_for_port(agwpe_port_b)

        yield {
            "agwpe_port_b": agwpe_port_b,
            "kiss_port_a": kiss_port_a,
            "kiss_pty_a": pty_path,
            "proc_a": proc_a,
            "proc_b": proc_b,
        }
    finally:
        kill_proc(proc_a)
        kill_proc(proc_b)
```

```python
def write_tncd_config(path, agwpe_port, kiss_type, kiss_host=None,
                      kiss_port=None, kiss_device=None, callsign="N0CALL-1"):
    """Write a tncd INI configuration file."""
    lines = [
        "[server]",
        "listen_host = 127.0.0.1",
        f"listen_port = {agwpe_port}",
        f"callsign = {callsign}",
        "",
        "[client]",
        f"type = {kiss_type}",
    ]
    if kiss_type == "tcp":
        lines.append(f"host = {kiss_host}")
        lines.append(f"port = {kiss_port}")

        lines.append(f"device = {kiss_device}")
        lines.append("baudrate = 9600")

        "",
        "[kiss]",
        "tx_delay = 10",
        "persistence = 255",
        "slot_time = 10",
        "tx_tail = 5",
        "full_duplex = 0",
    ])
```

```python
def read_direwolf_output(proc, timeout=15.0):
    """Read all available lines from Direwolf's stdout until timeout.

    Returns a list of decoded text lines.
    """
    lines = []
    deadline = time.monotonic() + timeout
    while time.monotonic() < deadline:
        line = proc.stdout.readline()
        if line:
            lines.append(line.decode(errors="replace").strip())
        else:
            time.sleep(0.1)
    return lines
```

```python
class TestAPRS:
    """Test APRS/UI frame forwarding through tncd."""

    def test_aprs_packets(self, tncd_instance, direwolf_pair):
        """Send APRS position, message, and telemetry through tncd,
        verify Direwolf-B decodes them."""
        from tests.emulator_agwpe import AGWPEClientEmulator

        agwpe_port = tncd_instance["agwpe_port"]
        proc_b = direwolf_pair["proc_b"]

        # Collect Direwolf-B output in a background thread
        output_lines = []
        stop_event = threading.Event()

        def reader():
            while not stop_event.is_set():
                line = proc_b.stdout.readline()
                if line:
                    output_lines.append(line.decode(errors="replace").strip())
                else:
                    time.sleep(0.05)

        reader_thread = threading.Thread(target=reader, daemon=True)
        reader_thread.start()

        try:
            # Send APRS packets through tncd
            async def send_packets():
                client = AGWPEClientEmulator("127.0.0.1", agwpe_port)
                await client.connect()

                # Register callsign first (X frame)
                from tests.emulator_agwpe import create_agwpe_frame
                reg = create_agwpe_frame(0, ord(b'X'), 'N0CALL-1', '', b'')
                client.writer.write(reg)
                await client.writer.drain()
                await client.receive(timeout=2.0)

                # Position report
                await client.send_unproto(
                    "N0CALL-1", "APRS",
                    b"!4903.50N/07201.75W-Test Position",
                )
                await asyncio.sleep(3.0)

                # Message
                await client.send_unproto(
                    "N0CALL-1", "APRS",
```

```python
def pat_compose_and_send(config_path, mbox_path, from_call, to_call,
                         subject, body, attachment_path, agwpe_addr,
                         timeout=180):
    """Compose a PAT message and send it via AGWPE connect.

    Args:
        config_path: Path to PAT config.json.
        mbox_path: Path to PAT mailbox directory.
        from_call: Sender callsign.
        to_call: Recipient callsign.
        subject: Message subject.
        body: Message body text.
        attachment_path: Path to file to attach (or None).
        agwpe_addr: Not used directly (already in config), kept for clarity.
        timeout: Seconds to wait for transfer.
    """
    pat_base = ["pat", "--config", config_path, "--mbox", mbox_path,
                "--mycall", from_call]

    # Compose the message
    compose_cmd = pat_base + [
        "compose",
        "--subject", subject,
        "--p2p-only",
    ]
    if attachment_path:
        compose_cmd.extend(["--attachment", str(attachment_path)])
    compose_cmd.append(to_call)

    result = subprocess.run(
        compose_cmd,
        input=body,
        capture_output=True, text=True, timeout=30,
    )
    if result.returncode != 0:
        raise RuntimeError(
            f"pat compose failed: {result.stderr}\n{result.stdout}"
        )

    # Connect and send
    connect_cmd = pat_base + [
        "--send-only",
        "connect", f"ax25+agwpe:///{to_call}",
    ]
    result = subprocess.run(
        connect_cmd,
        capture_output=True, text=True, timeout=timeout,
    )
    if result.returncode != 0:
        raise RuntimeError(
            f"pat connect failed: {result.stderr}\n{result.stdout}"
        )


def find_received_messages(mbox_path):
    """List message files in PAT's inbox.

    Returns list of Path objects for each .b2f or message file found.
    """
    inbox = Path(mbox_path) / "in"
    if not inbox.exists():
        return []
    return sorted(inbox.iterdir())
```

## Implementation Notes

### PipeWire Audio Troubleshooting

The trickiest part of this implementation is getting PipeWire audio routing to work correctly with Direwolf. The plan assumes Direwolf accepts PipeWire node names as ADEVICE arguments. If this doesn't work:

**Fallback approach:** Use `ADEVICE pipewire` (default) for both Direwolf instances, then after startup use `pw-link` to cross-connect their audio nodes. Identify nodes by PID:

```bash
# Find Direwolf-A's output node and Direwolf-B's input node
pw-link -o | grep <pid_a>  # output ports
pw-link -i | grep <pid_b>  # input ports
# Cross-link them
pw-link <output_port_a> <input_port_b>
pw-link <output_port_b> <input_port_a>
```

This would replace the `pw-loopback` fixture with a `pw-link`-based approach.

### PAT CLI Troubleshooting

If `pat --config` and `--mbox` flags don't work as expected, the alternative is to override `HOME` for each PAT subprocess:

```python
env = os.environ.copy()
env["HOME"] = str(tmp_path / "pat-a")
subprocess.run(["pat", "compose", ...], env=env, ...)
```

Then place config at `$HOME/.config/pat/config.json` and mailbox at `$HOME/.local/share/pat/mailbox/`.

### Timing at 1200 Baud

A 10KB attachment through AX.25 at 1200 baud:
- ~40 I-frames (256 bytes each)
- Each frame ≈ 2 seconds TX time + ack cycle
- Total ≈ 2-4 minutes per direction
- 180 second timeout per `pat connect` should be sufficient

The minimized TXDELAY/TXTAIL/SLOTTIME values in the Direwolf configs (10/5/10 vs typical 50/10/30) will help speed things up since there's no actual RF path or PTT switching.
