# AX.25 v2.2 (Phase 3) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add AX.25 v2.2 to the Go port — SABME/mod-128 with SABM/mod-8 auto-fallback, passive XID negotiation, and tolerance of received single-SREJ — validated against Direwolf.

**Architecture:** The KISS TNC is byte-transparent, so v2.2 is entirely host-side. The frame codec gains a `Modulo` field (8 or 128; mod-128 uses 2-byte I/S control fields). The l2 engine generalizes its `% 8` sequence arithmetic to `% modulo`, selects SABME vs SABM per-port, falls back to mod-8 on DM/FRMR/timeout exactly as Direwolf does, and answers XID commands with a min-negotiated response advertising SREJ-off. The bridge decodes inbound I/S control fields in two stages (address+type first, then re-decode at the link's modulo).

**Tech Stack:** Go 1.x, standard library only for the codec; existing `ax25`, `ax25/l2`, `internal/config`, `internal/bridge` packages; pytest for e2e.

## Global Constraints

- Reference implementation is **Direwolf 1.8.1** (`ax25_link.c`, `xid.c`). Local source copy: `/storage/W0NE-BBS-Backup/SD/home/pi/direwolf/src/`. This is greenfield — there is **no `tncd.py` counterpart** to mirror.

- **All existing tests must continue to pass** — mod-8 is the default and existing behavior must be byte-for-byte unchanged. Run `go test ./...` after every task.
- Sequence-number fields stay `uint8` (holds 0–127). Modulo is `uint8` (8 or 128). `(a-b) % m` on uint8 is correct for both moduli because 256 is a multiple of both.
- `ax25_version` config values are exactly `2.0` and `2.2` (strings); default `2.2`. Internally represented as `PortParams.AX25Version int` = `20` or `22`.
- Fallback counts replicate Direwolf: `maxV22 = N2Retry / 3` (integer division; `= 3` at default `N2Retry = 10`). First SABME is transmission try 1. Give up at `N2Retry` total tries.
- N1 (max I-field) default = **256 bytes**; mod-128 window clamp = `1..63`; mod-8 window clamp = `1..7`; default window = `3`.
- We **decode** SREJ and honor a received single-SREJ, but never **send** SREJ and never buffer out-of-order frames (that is phase 3.5).
- Branch: `feature/ax25-v22` (already created off `main`). Commit after every task with the message shown in that task's final step.

```go
package ax25

import "fmt"

// SREJMode is the selective-reject capability advertised in XID.
type SREJMode int

const (
	SREJNone SREJMode = iota // REJ only
	SREJSingle
	SREJMulti
)

// XIDParams holds the subset of AX.25 2.2 XID parameters tncd negotiates.
// Zero value of an omitted numeric field means "not present".
type XIDParams struct {
	FullDuplex       bool
	SREJ             SREJMode
	Modulo           int // 8 or 128
	IFieldLenRxBytes int // N1 in bytes (XID carries bits); 0 = omit
	WindowRx         int // 0 = omit
}

// XID format identifiers and parameter IDs (Direwolf xid.c).
const (
	xidFI = 0x82
	xidGI = 0x80

	xidPIClasses = 2
	xidPIHDLC    = 3
	xidPIIFieldRx = 6
	xidPIWindowRx = 8
)

// Classes-of-Procedures bits.
const (
	pvBalancedABM = 0x0100
	pvHalfDuplex  = 0x2000
	pvFullDuplex  = 0x4000
)

// HDLC-Optional-Functions bits (24-bit field).
const (
	pvREJ       = 0x020000
	pvSREJ      = 0x040000
	pvMultiSREJ = 0x000020
	pvExtAddr   = 0x800000
	pvModulo8   = 0x000400
	pvModulo128 = 0x000800
	pvTEST      = 0x002000
	pv16bitFCS  = 0x008000
	pvSyncTx    = 0x000002
)

// Encode builds the XID info field. command selects the command form (a "menu"
// of acceptable SREJ choices) vs the response form (a single SREJ bit), matching
// Direwolf's xid_encode. tncd only sends responses, but both forms are supported.
func (p XIDParams) Encode(command bool) []byte {
	var body []byte

	// Classes of Procedures (always present, PL=2).
	classes := pvBalancedABM
	if p.FullDuplex {
		classes |= pvFullDuplex
	} else {
		classes |= pvHalfDuplex
	}
	body = append(body, xidPIClasses, 2, byte(classes>>8), byte(classes))

	// HDLC Optional Functions (always present, PL=3).
	hdlc := pvExtAddr | pvTEST | pv16bitFCS | pvSyncTx
	if command {
		switch p.SREJ {
		case SREJSingle:
			hdlc |= pvREJ | pvSREJ
		case SREJMulti:
			hdlc |= pvREJ | pvSREJ | pvMultiSREJ
		default:
			hdlc |= pvREJ
		}
	} else {
		switch p.SREJ {
		case SREJSingle:
			hdlc |= pvSREJ
		case SREJMulti:
			hdlc |= pvMultiSREJ
		default:
			hdlc |= pvREJ
		}
	}
	if p.Modulo == 128 {
		hdlc |= pvModulo128
	} else {
		hdlc |= pvModulo8
	}
	body = append(body, xidPIHDLC, 3, byte(hdlc>>16), byte(hdlc>>8), byte(hdlc))

	// I Field Length Rx (bits), PL=2.
	if p.IFieldLenRxBytes > 0 {
		bits := p.IFieldLenRxBytes * 8
		body = append(body, xidPIIFieldRx, 2, byte(bits>>8), byte(bits))
	}
	// Window Size Rx, PL=1.
	if p.WindowRx > 0 {
		body = append(body, xidPIWindowRx, 1, byte(p.WindowRx))
	}

	out := []byte{xidFI, xidGI, byte(len(body) >> 8), byte(len(body))}
	return append(out, body...)
}

// ParseXID decodes an XID info field into the parameters tncd cares about.
func ParseXID(info []byte) (XIDParams, error) {
	p := XIDParams{Modulo: 8, SREJ: SREJNone}
	if len(info) < 4 || info[0] != xidFI || info[1] != xidGI {
		return p, fmt.Errorf("ax25: bad XID header")
	}
	glen := int(info[2])<<8 | int(info[3])
	body := info[4:]
	if glen > len(body) {
		return p, fmt.Errorf("ax25: XID group length overflow")
	}
	body = body[:glen]
	for len(body) >= 2 {
		pi, pl := body[0], int(body[1])
		if 2+pl > len(body) {
			return p, fmt.Errorf("ax25: XID param overflow")
		}
		pv := body[2 : 2+pl]
		switch pi {
		case xidPIClasses:
			if pl == 2 {
				x := int(pv[0])<<8 | int(pv[1])
				p.FullDuplex = x&pvFullDuplex != 0
			}
		case xidPIHDLC:
			if pl == 3 {
				x := int(pv[0])<<16 | int(pv[1])<<8 | int(pv[2])
				if x&pvModulo128 != 0 {
					p.Modulo = 128
				}
				switch {
				case x&pvMultiSREJ != 0:
					p.SREJ = SREJMulti
				case x&pvSREJ != 0:
					p.SREJ = SREJSingle
				default:
					p.SREJ = SREJNone
				}
			}
		case xidPIIFieldRx:
			if pl == 2 {
				bits := int(pv[0])<<8 | int(pv[1])
				p.IFieldLenRxBytes = bits / 8
			}
		case xidPIWindowRx:
			if pl == 1 {
				p.WindowRx = int(pv[0])
			}
		}
		body = body[2+pl:]
	}
	return p, nil
}
```

- `dispatchI`: `gap := (f.NS - expectedNS) % c.modulo`; `c.recvSeq = (f.NS + 1) % c.modulo`
- `retransmitFrom`: `seq = (seq + 1) % c.modulo`

Frames must carry the right `Modulo` so `Bytes()` emits the correct control
width. Two places matter, because `drainOutbound` and `retransmitFrom` call
`f.Bytes()` to fill `retransmitBuf` **before** the frame reaches `sendFrame` —
so those literals need `Modulo` set inline:

1. In `drainOutbound`, add `Modulo: c.modulo,` to the `f := &ax25.Frame{...}`
   I-frame literal (before `raw := f.Bytes()`).
2. In `retransmitFrom`, add `Modulo: c.modulo,` to the rebuilt `f := &ax25.Frame{...}`
   literal, and change its internal re-parse to honor the stored modulo:

```go
	orig, err := ax25.ParseModulo(raw, int(c.modulo))
```

3. For every other I/S frame (RR/REJ/RNR responses and RR polls built via
   `respFrame`/`cmdFrame` or inline literals, which do **not** store raw), stamp
   the modulo in one central place — `sendFrame` — so no call site is missed. An
   outgoing frame's `Src` is our local call and `Dst` is the remote, which is
   exactly the connection key, so `ModuloFor` resolves it:

```go
func (t *Table) sendFrame(port int, f *ax25.Frame) {
	if f.Modulo == 0 && (f.Type == ax25.I || f.Type.IsS()) {
		f.Modulo = uint8(t.ModuloFor(port, f.Src.String(), f.Dst.String()))
	}
	if t.hooks.SendAX25 != nil {
		t.hooks.SendAX25(port, f)
	}
}
```

(U-frames keep `Modulo == 0`, which `Bytes()` treats as mod-8 — correct, since
U-frames are 1-byte regardless. A mod-8 conn resolves to `Modulo = 8`, also
1-byte. Only mod-128 I/S frames get the 2-byte control.)

Add the `ModuloFor` method:

```go
// ModuloFor returns the negotiated modulo (8 or 128) for the connection
// addressed by (port, local, remote), or 8 if there is no such connection.
// The bridge uses this to decode inbound I/S control fields at the right width.
func (t *Table) ModuloFor(port int, local, remote string) int {
	if c := t.Get(port, local, remote); c != nil {
		return int(c.modulo)
	}
	return 8
}
```

```go
// dispatchXID answers an XID command with a min-negotiated XID response.
// tncd is a passive responder: it never initiates XID. We always advertise
// SREJ off (which is what turns Direwolf's default-on single-SREJ off) and
// our fixed N1 = 256, negotiating the window down to the peer's request.
// Mirrors Direwolf xid_frame (mdl_state_0_ready, command branch).
func (t *Table) dispatchXID(port int, f *ax25.Frame, src, dst string) {
	if !f.Command || !f.PF {
		return // only XID command with P=1 is actionable; a response is unexpected
	}
	if !t.isLocal(port, dst) {
		return
	}
	c := t.Get(port, dst, src)
	if c == nil {
		return
	}
	their, err := ax25.ParseXID(f.Info)
	if err != nil {
		return
	}
	pp := t.portParams(port)
	window := pp.MaxWindow
	if their.WindowRx > 0 && their.WindowRx < window {
		window = their.WindowRx
	}
	resp := ax25.XIDParams{
		FullDuplex:       false,
		SREJ:             ax25.SREJNone,
		Modulo:           128,
		IFieldLenRxBytes: 256,
		WindowRx:         window,
	}
	out := respFrame(src, dst, c.Via, ax25.XID, true) // response, F=1
	out.Info = resp.Encode(false)
	t.sendFrame(port, out)
}
```
